# 「诗芽」Android TV 应用实现计划

> **面向 AI 代理的工作者:** 必需子技能:使用 superpowers:subagent-driven-development(推荐)或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框(`- [ ]`)语法来跟踪进度。

**目标:** 构建「诗芽」Android TV 应用:画卷雅韵视觉、5 屏导航、KTV 式朗读、跟读模式、线装书讲解、收藏,全离线运行于 Android 10(API 29)。

**架构:** 单模块 MVVM:Compose TV UI ← ViewModel ← Repository/朗读引擎 ← 内置 assets(JSON 数据 + mp3 音频 + 逐字时间轴)。依赖内容管线产物;管线未完成前用本计划自带的样例资产开发。

**技术栈:** Kotlin 2.0 + Compose TV(tv-material 1.0)+ Media3 ExoPlayer + kotlinx-serialization + DataStore + Navigation Compose。minSdk 24,targetSdk 34。

## 风险门(任务 1 前确认)

1. **JDK 17:** 按顺序找:`/usr/libexec/java_home -V` → `/Applications/Android Studio.app/Contents/jbr/Contents/Home` → 华为云镜像安装(命令见任务 1)。任一路径可用即可。
2. **Android SDK 下载:** 优先 `dl.google.com`;超时则用腾讯镜像 `https://mirrors.cloud.tencent.com/AndroidSDK/`(命令见任务 1)。
3. **Gradle 发行版:** 优先 `services.gradle.org`;失败则改 `gradle-wrapper.properties` 为腾讯镜像 `https://mirrors.cloud.tencent.com/gradle/`。
4. **Maven 依赖:** 一律走阿里云镜像(settings.gradle.kts 中配置),`google()` 原始源在国内不可靠。
5. **霞鹜文楷字体:** 从 npmmirror 拉取;失败则降级为系统字体(楷体 fallback 栈),不阻塞开发。

```kotlin
// PoemRepository.kt
class PoemRepository(private val context: Context) {
    private val json = Json { ignoreUnknownKeys = true }
    private val basePath = "assets"      // 生产;样例开发期用 "assets/sample" 的 fallback 逻辑见下

    suspend fun loadIndex(): PoemIndex = withContext(Dispatchers.IO) {
        context.assets.open("$basePath/index.json").use { json.decodeFromString(it.readBytes().decodeToString()) }
    }

    suspend fun loadPoem(id: String): Poem = withContext(Dispatchers.IO) {
        context.assets.open("$basePath/poems/$id.json").use { json.decodeFromString(it.readBytes().decodeToString()) }
    }

    suspend fun loadTiming(id: String): PoemTiming? = withContext(Dispatchers.IO) {
        runCatching {
            context.assets.open("$basePath/timing/$id.json").use { json.decodeFromString(it.readBytes().decodeToString()) }
        }.getOrNull()
    }

    fun audioUri(id: String): String = "file:///android_asset/$basePath/audio/$id.mp3"

    fun hasAudio(id: String): Boolean = runCatching {
        context.assets.openFd("$basePath/audio/$id.mp3").use { it.length > 0 }
    }.getOrDefault(false)
}
```

```kotlin
// FavoritesStore.kt
private val Context.dataStore by preferencesDataStore(name = "shiya_prefs")

class FavoritesStore(private val context: Context) {
    val favorites: Flow<Set<String>> = context.dataStore.data.map { it[FAVORITES] ?: emptySet() }

    suspend fun toggle(id: String) {
        context.dataStore.edit { p ->
            val cur = p[FAVORITES] ?: emptySet()
            p[FAVORITES] = if (id in cur) cur - id else cur + id
        }
    }

    suspend fun isFavorite(id: String): Boolean = favorites.first().contains(id)

    companion object { private val FAVORITES = stringSetPreferencesKey("favorites") }
}
```

// TimingModel.kt
data class CharPosition(val line: Int, val char: Int, val progress: Float)

object TimingModel {
    /** 查询播放位置对应的 (行, 字, 字内进度)。句间间隙停在上一句末字满进度。 */
    fun query(timing: PoemTiming, positionMs: Long): CharPosition {
        val chars = timing.chars
        if (chars.isEmpty()) return CharPosition(0, 0, 1f)
        var last = chars.first()
        for (c in chars) {
            if (positionMs < c.startMs) break
            last = c
        }
        val dur = last.endMs - last.startMs
        val progress = if (dur <= 0) 1f
            else ((positionMs - last.startMs).toFloat() / dur).coerceIn(0f, 1f)
        return CharPosition(last.line, last.index, progress)
    }

    fun lineStart(timing: PoemTiming, line: Int): Long =
        timing.chars.firstOrNull { it.line == line }?.startMs ?: 0L

    fun lineEnd(timing: PoemTiming, line: Int): Long =
        timing.chars.lastOrNull { it.line == line }?.endMs ?: 0L

    fun lineCount(timing: PoemTiming): Int = (timing.chars.lastOrNull()?.line ?: 0) + 1
}

```kotlin
// RecitationEngine.kt
enum class Speed(val multiplier: Float, val label: String) {
    X06(0.6f, "慢"), X08(0.8f, "中"), X10(1.0f, "常");
    fun next(): Speed = when (this) { X06 -> X08; X08 -> X10; X10 -> X06 }
}

data class ReaderState(
    val playing: Boolean, val positionMs: Long, val durationMs: Long,
    val speed: Speed, val followMode: Boolean,
    val current: CharPosition, val audioAvailable: Boolean,
    val lineEndReached: Boolean = false,
) {
    companion object {
        fun reduce(s: ReaderState, a: ReaderAction): ReaderState = when (a) {
            ReaderAction.TogglePlay -> s.copy(playing = !s.playing, lineEndReached = false)
            ReaderAction.CycleSpeed -> s.copy(speed = s.speed.next())
            ReaderAction.ToggleFollow -> s.copy(followMode = !s.followMode)
            is ReaderAction.PositionChanged -> {
                val overLineEnd = s.followMode && a.lineEnd >= 0 && a.positionMs >= a.lineEnd && s.playing
                s.copy(positionMs = a.positionMs, current = a.current,
                       playing = if (overLineEnd) false else s.playing,
                       lineEndReached = overLineEnd)
            }
            is ReaderAction.Seeked -> s.copy(positionMs = a.positionMs, current = a.current, lineEndReached = false)
            is ReaderAction.Loaded -> s.copy(positionMs = 0, durationMs = a.durationMs,
                                             current = CharPosition(0, 0, 0f), playing = true,
                                             audioAvailable = true, lineEndReached = false)
            ReaderAction.Unavailable -> s.copy(audioAvailable = false, playing = false)
            ReaderAction.Ended -> s.copy(playing = false, positionMs = s.durationMs)
        }
    }
}

sealed interface ReaderAction {
    data object TogglePlay : ReaderAction
    data object CycleSpeed : ReaderAction
    data object ToggleFollow : ReaderAction
    data class PositionChanged(val positionMs: Long, val current: CharPosition, val lineEnd: Long) : ReaderAction
    data class Seeked(val positionMs: Long, val current: CharPosition) : ReaderAction
    data class Loaded(val durationMs: Long) : ReaderAction
    data object Unavailable : ReaderAction
    data object Ended : ReaderAction
}

class RecitationEngine(private val context: Context) {
    private var player: ExoPlayer? = null
    private var timing: PoemTiming? = null
    private val _state = MutableStateFlow(ReaderState(
        playing = false, positionMs = 0, durationMs = 0, speed = Speed.X08,
        followMode = false, current = CharPosition(0, 0, 0f), audioAvailable = true))
    val state: StateFlow<ReaderState> = _state

    private val poller = CoroutineScope(SupervisorJob() + Dispatchers.Default)

    fun load(audioUri: String, timing: PoemTiming?) {
        this.timing = timing
        if (timing == null) { _state.value = ReaderState.reduce(_state.value, ReaderAction.Unavailable); return }
        val p = ExoPlayer.Builder(context).build().apply {
            setMediaItem(MediaItem.fromUri(audioUri))
            prepare()
            playWhenReady = true
            addListener(object : Player.Listener {
                override fun onPlaybackStateChanged(state: Int) {
                    if (state == Player.STATE_ENDED) _state.value = ReaderState.reduce(_state.value, ReaderAction.Ended)
                }
            })
        }
        player?.release(); player = p
        _state.value = ReaderState.reduce(_state.value, ReaderAction.Loaded(p.duration))
        startPolling()
    }

    private fun startPolling() {
        poller.coroutineContext[Job]?.cancel()
        poller.launch {
            while (isActive) {
                val p = player ?: break
                val pos = p.currentPosition
                val t = timing ?: break
                val cur = TimingModel.query(t, pos)
                val lineEnd = TimingModel.lineEnd(t, cur.line)
                _state.value = ReaderState.reduce(_state.value, ReaderAction.PositionChanged(pos, cur, lineEnd))
                delay(32)
            }
        }
    }

    fun togglePlay() {
        val s = _state.value
        _state.value = ReaderState.reduce(s, ReaderAction.TogglePlay)
        player?.playWhenReady = _state.value.playing
    }

    fun cycleSpeed() {
        _state.value = ReaderState.reduce(_state.value, ReaderAction.CycleSpeed)
        player?.setPlaybackParameters(PlaybackParameters(_state.value.speed.multiplier))
    }

    fun toggleFollow() { _state.value = ReaderState.reduce(_state.value, ReaderAction.ToggleFollow) }

    fun seekToLine(line: Int) {
        val t = timing ?: return
        val start = TimingModel.lineStart(t, line)
        player?.seekTo(start)
        _state.value = ReaderState.reduce(_state.value,
            ReaderAction.Seeked(start, TimingModel.query(t, start)))
        player?.playWhenReady = _state.value.playing
    }

    fun release() { player?.release(); player = null; poller.coroutineContext[Job]?.cancel() }
}
```

enum class LibraryMode { Difficulty, Poet, Favorites }

data class LibraryUiState(
    val loading: Boolean = true,
    val groups: List<Pair<String, List<IndexEntry>>> = emptyList(),  // 组名 → 诗词列表
    val error: String? = null,
)

class LibraryViewModel(
    private val repository: PoemRepository,
    private val favorites: FavoritesStore,
    private val mode: LibraryMode,
) : ViewModel() {
    val uiState = MutableStateFlow(LibraryUiState())
    init { refresh() }
    fun refresh() {
        viewModelScope.launch {
            uiState.value = LibraryUiState(loading = true)
            runCatching {
                val idx = repository.loadIndex()
                val groups = when (mode) {
                    LibraryMode.Difficulty -> listOf(
                        "启蒙篇" to (idx.byDifficulty["qimeng"] ?: emptyList()),
                        "进阶篇" to (idx.byDifficulty["jinjie"] ?: emptyList()),
                        "小学必背" to (idx.byDifficulty["bibei"] ?: emptyList()),
                    ).filter { it.second.isNotEmpty() }
                    LibraryMode.Poet -> idx.byPoet.map { (poet, ids) ->
                        poet to ids.map { idx.poems[it]!! } }.sortedBy { it.first }
                    LibraryMode.Favorites -> {
                        val favIds = favorites.favorites.first()
                        listOf("我的收藏" to favIds.mapNotNull { idx.poems[it] })
                    }
                }.map { (name, entries) -> name to entries }
                uiState.value = LibraryUiState(loading = false, groups = groups)
            }.onFailure { uiState.value = LibraryUiState(loading = false, error = it.message) }
        }
    }
}

```kotlin
class ReaderViewModel(
    private val poemId: String,
    private val repository: PoemRepository,
    private val engine: RecitationEngine,
    private val favorites: FavoritesStore,
) : ViewModel() {
    val poem = MutableStateFlow<Poem?>(null)
    val engineState = engine.state
    val isFavorite = MutableStateFlow(false)

    init {
        viewModelScope.launch {
            val p = runCatching { repository.loadPoem(poemId) }.getOrNull()
            poem.value = p
            if (p != null) {
                val timing = repository.loadTiming(poemId)
                val hasAudio = repository.hasAudio(poemId)
                if (hasAudio) engine.load(repository.audioUri(poemId), timing)
                else engine.load("", null)   // 进入 Unavailable
            }
            isFavorite.value = favorites.isFavorite(poemId)
        }
    }

    fun togglePlay() = engine.togglePlay()
    fun cycleSpeed() = engine.cycleSpeed()
    fun toggleFollow() = engine.toggleFollow()
    fun seekToLine(line: Int) = engine.seekToLine(line)
    fun toggleFavorite() {
        viewModelScope.launch {
            favorites.toggle(poemId)
            isFavorite.value = favorites.isFavorite(poemId)
        }
    }

    override fun onCleared() { engine.release() }
}
```
